-- Actividad semana 9. Ejecutar en una base exclusiva de pruebas.
-- Las validaciones son decisiones de diseño, no reglas textuales del PDF.
CREATE TABLE dbo.Productos (
 IdProducto INT PRIMARY KEY IDENTITY,
 Nombre NVARCHAR(100), Stock INT, Precio DECIMAL(10,2),
 CONSTRAINT CK_Productos_Nombre CHECK (Nombre IS NOT NULL AND LEN(LTRIM(RTRIM(Nombre))) > 0),
 CONSTRAINT CK_Productos_Stock CHECK (Stock IS NOT NULL AND Stock >= 0),
 CONSTRAINT CK_Productos_Precio CHECK (Precio IS NOT NULL AND Precio >= 0)
);
CREATE TABLE dbo.AuditoriaInventario (
 IdAuditoria INT PRIMARY KEY IDENTITY,
 IdProducto INT, Accion NVARCHAR(10), Fecha DATETIME DEFAULT GETDATE()
);
-- Sin FK a Productos: la auditoría debe conservar las bajas.

CREATE PROCEDURE dbo.sp_AltaProducto
 @Nombre NVARCHAR(100), @Stock INT, @Precio DECIMAL(10,2)
AS
BEGIN
 SET NOCOUNT ON;
 BEGIN TRY
  IF @Nombre IS NULL OR LEN(LTRIM(RTRIM(@Nombre)))=0
   THROW 50001, 'El nombre no puede estar vacío.', 1;
  IF @Stock IS NULL OR @Stock < 0 THROW 50002, 'El stock debe ser no negativo.', 1;
  IF @Precio IS NULL OR @Precio < 0 THROW 50003, 'El precio debe ser no negativo.', 1;
  INSERT dbo.Productos(Nombre,Stock,Precio) VALUES(@Nombre,@Stock,@Precio);
  PRINT 'Producto insertado correctamente.';
 END TRY
 BEGIN CATCH
  PRINT 'Error al insertar el producto: ' + ERROR_MESSAGE();
  THROW;
 END CATCH
END;

CREATE PROCEDURE dbo.sp_BajaProducto @IdProducto INT
AS
BEGIN
 SET NOCOUNT ON;
 BEGIN TRY
  IF @IdProducto IS NULL OR @IdProducto <= 0
   THROW 50004, 'El identificador debe ser positivo.', 1;
  -- DELETE y @@ROWCOUNT evitan separar la comprobación de la operación.
  DELETE dbo.Productos WHERE IdProducto=@IdProducto;
  IF @@ROWCOUNT=0 PRINT 'Producto no existe.';
  ELSE PRINT 'Producto eliminado correctamente.';
 END TRY
 BEGIN CATCH
  PRINT 'Error al eliminar el producto: ' + ERROR_MESSAGE();
  THROW;
 END CATCH
END;

-- Ampliación propuesta para cubrir menciones de la rúbrica.
-- INSTEAD OF reemplaza la operación; no es un trigger BEFORE.
CREATE TRIGGER dbo.trg_Validar_Actualizacion ON dbo.Productos INSTEAD OF UPDATE
AS
BEGIN
 SET NOCOUNT ON;
 IF EXISTS(SELECT 1 FROM inserted WHERE Nombre IS NULL OR LEN(LTRIM(RTRIM(Nombre)))=0
  OR Stock IS NULL OR Stock<0 OR Precio IS NULL OR Precio<0)
  THROW 50005, 'Actualización rechazada: nombre, stock o precio inválidos.', 1;
 UPDATE p SET Nombre=i.Nombre, Stock=i.Stock, Precio=i.Precio
 FROM dbo.Productos p JOIN inserted i ON p.IdProducto=i.IdProducto;
END;

CREATE TRIGGER dbo.trg_Auditoria_ActualizarProducto ON dbo.Productos AFTER UPDATE
AS
BEGIN
 SET NOCOUNT ON;
 -- Condicional: no registrar una actualización sin cambios efectivos.
 INSERT dbo.AuditoriaInventario(IdProducto,Accion)
 SELECT i.IdProducto,'UPD' FROM inserted i JOIN deleted d ON i.IdProducto=d.IdProducto
 WHERE i.Nombre<>d.Nombre OR i.Stock<>d.Stock OR i.Precio<>d.Precio;
END;

CREATE PROCEDURE dbo.sp_ConsultarInventarioCursor
AS
BEGIN
 SET NOCOUNT ON;
 DECLARE @Id INT,@Nombre NVARCHAR(100),@Stock INT;
 -- Demostración académica de recorrido. Para listar inventario se prefiere SELECT.
 DECLARE inventario CURSOR LOCAL FAST_FORWARD FOR
 SELECT IdProducto,Nombre,Stock FROM dbo.Productos ORDER BY IdProducto;
 BEGIN TRY
  OPEN inventario;
  FETCH NEXT FROM inventario INTO @Id,@Nombre,@Stock;
  WHILE @@FETCH_STATUS=0
  BEGIN
   PRINT CONCAT('ID=',@Id,'; Nombre=',@Nombre,'; Stock=',@Stock);
   FETCH NEXT FROM inventario INTO @Id,@Nombre,@Stock;
  END;
  CLOSE inventario;
  DEALLOCATE inventario;
 END TRY
 BEGIN CATCH
  IF CURSOR_STATUS('local','inventario')>=0 CLOSE inventario;
  IF CURSOR_STATUS('local','inventario')>=-1 DEALLOCATE inventario;
  THROW;
 END CATCH
END;
